import requests

BASE_URL = "https://lk.rt.ru/api/auth/by-account"


def test_auth_by_account_success():
    payload = {
        "account": "1234567890",
        "password": "correct_password"
    }

    response = requests.post(BASE_URL, json=payload)

    assert response.status_code == 200, "AUTH_200_EXPECTED"

    response_json = response.json()
    assert "redirect_uri" in response_json, "REDIRECT_URI_NOT_FOUND"

#5. Неверный пароль
def test_auth_by_account_wrong_password():
    payload = {
        "account": "1234567890",
        "password": "wrong_password"
    }

    response = requests.post(BASE_URL, json=payload)

    assert response.status_code == 401, "AUTH_401_WRONG_PASSWORD"

    response_json = response.json()
    assert response_json.get("message") == "Неверный логин или пароль", \
        "ERROR_MESSAGE_INCORRECT"

def test_auth_by_account_invalid_format():
