#Сценарий авторизации клиента по номеру телефона, кнопка "Логин"
"Сценарий авторизации (логика) 

Актор: Клиент

Точка входа: кнопка «Логин»

Тип логина: номер телефона / email / логин (обобщённо)


Основной сценарий (Happy Path) 

Клиент вводит логин + пароль


Нажимает кнопку «Логин»


Система:


валидирует формат логина
проверяет связку Логин + Пароль


При успехе:


находит УЗ (учётную запись)


аутентифицирует клиента


создаёт сессию / JWT

 
редиректит на redirect_uri


Негативный сценарий


При некорректной связке:

сообщение: «Неверный логин или пароль»


элемент «Забыл пароль» подсвечивается оранжевым"

#JavaScript
document.getElementById("loginForm").addEventListener("submit", async (e) => {
  e.preventDefault();

  const login = document.getElementById("login").value.trim();
  const password = document.getElementById("password").value;
  const errorBlock = document.getElementById("error");
  const forgotLink = document.getElementById("forgotPassword");

  errorBlock.classList.add("hidden");
  forgotLink.classList.remove("forgot-orange");

  // Клиентская валидация
  if (!login || !password) {
    return;
  }

  try {
    const response = await fetch("/api/auth/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        login,
        password,
        redirect_uri: window.location.search
      })
    });

    if (!response.ok) {
      throw new Error("AUTH_FAILED");
    }

    const data = await response.json();
    window.location.href = data.redirect_uri;

  } catch (err) {
    errorBlock.classList.remove("hidden");
    forgotLink.classList.add("forgot-orange");
  }
});

#Backend-логика (псевдокод)
function login(request):
    login = request.login
    password = request.password

    if not isValidLoginFormat(login):
        return 401

    user = findUserByLogin(login)

    if user == null:
        return 401

    if not verifyPassword(password, user.password_hash):
        return 401

    session = createSession(user.id)

    return 200 {
        redirect_uri: request.redirect_uri
    }
