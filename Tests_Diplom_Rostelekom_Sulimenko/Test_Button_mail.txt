#JavaScript
document.getElementById("emailLoginForm").addEventListener("submit", (e) => {
  e.preventDefault();

  const email = document.getElementById("email").value.trim();
  const password = document.getElementById("password").value;
  const error = document.getElementById("error");
  const forgot = document.getElementById("forgotPassword");

  error.classList.add("hidden");
  forgot.classList.remove("forgot-orange");

  // Проверка формата почты
  const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  if (!emailPattern.test(email)) {
    showError();
    return;
  }

  // Проверка связки
  if (email !== "test@mail.ru" || password !== "123456") {
    showError();
    return;
  }

  // Редирект
  window.location.href = "/redirect_uri";

  function showError() {
    error.classList.remove("hidden");
    forgot.classList.add("forgot-orange");
  }
});

#Backend
function loginByEmail(request):
    email = request.email
    password = request.password

    if not validEmailFormat(email):
        return AUTH_ERROR

    user = findUserByEmail(email)

    if user is null:
        return AUTH_ERROR

    if password != user.password:
        return AUTH_ERROR

    authenticate(user)

    redirect(request.redirect_uri)
